cmake_minimum_required(VERSION 3.20)
project(gbrt_portability_tests LANGUAGES C CXX)
enable_testing()
add_executable(gb_filesystem_test filesystem_test.cpp ../../src/gb_filesystem.cpp)
target_include_directories(gb_filesystem_test PRIVATE ../../include)
add_executable(gb_network_test network_test.cpp
    ../../src/serial_link.cpp ../../src/network_discovery.cpp)
target_include_directories(gb_network_test PRIVATE ../../include ../../src)
find_package(Threads REQUIRED)
target_link_libraries(gb_network_test PRIVATE Threads::Threads)
if(WIN32)
    target_link_libraries(gb_network_test PRIVATE ws2_32)
endif()
add_executable(gb_string_test string_test.c)
target_include_directories(gb_string_test PRIVATE ../../src)
target_compile_features(gb_string_test PRIVATE c_std_11)
foreach(target gb_filesystem_test gb_network_test gb_string_test)
    set_target_properties(${target} PROPERTIES
        CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
    if(MSVC)
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
        target_compile_options(${target} PRIVATE /W4 /WX $<$<COMPILE_LANGUAGE:CXX>:/permissive->)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror $<$<COMPILE_LANGUAGE:CXX>:-Werror=c++20-extensions>)
    endif()
endforeach()
add_test(NAME host_filesystem COMMAND gb_filesystem_test)
add_test(NAME host_strings COMMAND gb_string_test)
add_test(NAME host_link COMMAND gb_network_test link)
add_test(NAME host_discovery COMMAND gb_network_test discovery)
set_tests_properties(host_link host_discovery PROPERTIES TIMEOUT 40)

if(MSVC)
    # Existing GBContext register pairs use anonymous structs in a union.
    target_compile_options(gb_network_test PRIVATE /wd4201)
endif()
