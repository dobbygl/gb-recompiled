cmake_minimum_required(VERSION 3.20)
project(gbrt_portability_tests LANGUAGES C CXX)
enable_testing()
add_executable(gb_filesystem_test filesystem_test.cpp ../../src/gb_filesystem.cpp)
target_include_directories(gb_filesystem_test PRIVATE ../../include)
set_target_properties(gb_filesystem_test PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES)
if(MSVC)
    target_compile_options(gb_filesystem_test PRIVATE /W4 /WX)
else()
    target_compile_options(gb_filesystem_test PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME host_filesystem COMMAND gb_filesystem_test)
